cmake_minimum_required(VERSION 3.18)
project(cvdaw LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CVDAW_TARGET "Build Linux ALSA engine and Rubber Band worker" OFF)
add_library(cvdaw INTERFACE)
target_include_directories(cvdaw INTERFACE include)
target_compile_options(cvdaw INTERFACE -Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)
add_executable(cvdaw-brain src/brain.cpp)
target_link_libraries(cvdaw-brain PRIVATE cvdaw Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
 target_link_libraries(cvdaw-brain PRIVATE rt)
endif()
if(CVDAW_TARGET)
 if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Target audio backend requires Linux")
 endif()
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(ALSA REQUIRED IMPORTED_TARGET alsa)
 pkg_check_modules(RUBBERBAND REQUIRED IMPORTED_TARGET rubberband)
 pkg_check_modules(SNDFILE REQUIRED IMPORTED_TARGET sndfile)
 add_executable(cvdaw-audio src/audio.cpp)
 target_link_libraries(cvdaw-audio PRIVATE cvdaw PkgConfig::ALSA Threads::Threads rt)
 add_executable(cvdaw-stems src/stems.cpp)
 target_link_libraries(cvdaw-stems PRIVATE cvdaw PkgConfig::RUBBERBAND PkgConfig::SNDFILE Threads::Threads rt)
endif()
enable_testing()
add_executable(cvdaw-tests tests/core.cpp)
target_link_libraries(cvdaw-tests PRIVATE cvdaw Threads::Threads)
add_test(NAME core COMMAND cvdaw-tests)

option(CVDAW_MODULES45 "Build the standalone miniaudio Modules 4 and 5" OFF)
if(CVDAW_MODULES45)
 add_subdirectory(realtime)
endif()
