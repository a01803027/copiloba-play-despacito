cmake_minimum_required(VERSION 3.18)
project(cv_daw_realtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(CV_DAW_CORTEX_A35 "Optimize for ARMv8-A Cortex-A35 (use AArch64 compiler)" OFF)
option(CV_DAW_FAST_MATH "Use requested fast-math DSP optimization" ON)
option(CV_DAW_BUILD_AUDIO "Build miniaudio engine (CoreAudio on macOS, ALSA on Linux)" ON)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
 set(CV_DAW_RB_DEFAULT ON)
else()
 set(CV_DAW_RB_DEFAULT OFF)
endif()
option(CV_DAW_WITH_RUBBERBAND "Enable real four-stem streaming time stretch" ${CV_DAW_RB_DEFAULT})
add_library(cv_daw_flags INTERFACE)
target_include_directories(cv_daw_flags INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(cv_daw_flags INTERFACE -Wall -Wextra -Wpedantic -O3 -ftree-vectorize)
if(CV_DAW_FAST_MATH)
 target_compile_options(cv_daw_flags INTERFACE -ffast-math)
endif()
if(CV_DAW_CORTEX_A35)
 if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  message(FATAL_ERROR "Cortex-A35 target requires AArch64; configure using the OpenSTLinux SDK toolchain")
 endif()
 target_compile_options(cv_daw_flags INTERFACE -mcpu=cortex-a35)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
 target_compile_definitions(cv_daw_flags INTERFACE _GNU_SOURCE)
 target_link_libraries(cv_daw_flags INTERFACE rt)
endif()
add_library(cv_daw_ipc SHARED ipc_bridge_c.cpp)
target_link_libraries(cv_daw_ipc PRIVATE cv_daw_flags Threads::Threads)
if(CV_DAW_BUILD_AUDIO)
 add_executable(cv-daw-engine audio_engine.cpp)
 target_link_libraries(cv-daw-engine PRIVATE cv_daw_flags Threads::Threads ${CMAKE_DL_LIBS})
 # Vendor header is kept intact; suppress warnings originating from its implementation.
 set_source_files_properties(audio_engine.cpp PROPERTIES COMPILE_OPTIONS "-Wno-unused-function")
 if(CV_DAW_WITH_RUBBERBAND)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CV_DAW_RB REQUIRED IMPORTED_TARGET rubberband>=3.0)
  target_compile_definitions(cv-daw-engine PRIVATE CV_DAW_WITH_RUBBERBAND=1)
  target_link_libraries(cv-daw-engine PRIVATE PkgConfig::CV_DAW_RB)
 endif()
endif()
enable_testing()
add_executable(cv-daw-rt-tests tests/rt_tests.cpp)
target_link_libraries(cv-daw-rt-tests PRIVATE cv_daw_flags Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
 target_sources(cv-daw-rt-tests PRIVATE tests/rt_guards.cpp)
 target_link_options(cv-daw-rt-tests PRIVATE
  -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--wrap=free
  -Wl,--wrap=pthread_mutex_lock -Wl,--wrap=pthread_mutex_trylock -Wl,--wrap=pthread_mutex_unlock)
endif()
add_test(NAME modules45 COMMAND cv-daw-rt-tests)

if(CV_DAW_BUILD_AUDIO AND NOT CMAKE_CROSSCOMPILING)
 find_package(Python3 COMPONENTS Interpreter)
 if(Python3_Interpreter_FOUND)
  add_test(NAME modules45_python_ipc COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_mock.py $<TARGET_FILE:cv-daw-engine> $<TARGET_FILE:cv_daw_ipc>)
  set_tests_properties(modules45_python_ipc PROPERTIES TIMEOUT 15 RUN_SERIAL TRUE)
 endif()
endif()

if(CV_DAW_BUILD_AUDIO AND Python3_Interpreter_FOUND AND NOT CMAKE_CROSSCOMPILING)
 add_test(NAME audition_wav COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_wav.py $<TARGET_FILE:cv-daw-engine> ${CMAKE_CURRENT_SOURCE_DIR}/../artifacts/dsp-audition.wav)
 set_tests_properties(audition_wav PROPERTIES RUN_SERIAL TRUE)
 add_test(NAME composer_rules COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_composer.py)
 set_tests_properties(composer_rules PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/..)
endif()
if(CV_DAW_BUILD_AUDIO AND Python3_Interpreter_FOUND AND NOT CMAKE_CROSSCOMPILING)
 add_test(NAME composer_ipc COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_composer_ipc.py $<TARGET_FILE:cv-daw-engine> $<TARGET_FILE:cv_daw_ipc>)
 set_tests_properties(composer_ipc PROPERTIES RUN_SERIAL TRUE TIMEOUT 20)
endif()
